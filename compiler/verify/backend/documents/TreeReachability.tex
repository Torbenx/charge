\begin{algorithmic}[1]
\Function{BuildLists}{$A$}
\State $L_1 \gets \text{Empty list}$
\State $L_2 \gets \text{Empty list}$
\For{child $C$ of $A$} \label{state:forLoop}
	\State $(T_1, T_2) \gets \Call{BuildLists}{C}$
	\State $L_1 \gets \Call{concat}{L_1, T_1}$
	\State $L_2 \gets \Call{concat}{T_2, L_2}$
\EndFor \label{state:endForLoop}
\State $L_1 \gets \Call{concat}{\{A\}, L_1}$ \label{state:finalAssign1}
\State $L_2 \gets \Call{concat}{\{A\}, L_2}$ \label{state:finalAssign2}
\State \Return $(L_1, L_2)$
\EndFunction
\end{algorithmic}
Let $I_1(A)$ and $I_2(A)$ be the position of $A$ in the first and second list returned by $\Call{BuildLists}{R}$ where $R$ is the root of the tree. Then $B$ is an (indirect) child of $A$ if and only if $I_1(B) > I_1(A)$ and $I_2(B) > I_2(A)$.

\begin{proof}
First if $B$ is a direct or indirect child of a $A$ then it is clear from lines \ref{state:finalAssign1} and \ref{state:finalAssign2} that $B$ appears in both lists after $A$. Thus $I_j(B) > I_j(A)$ for both lists.
\par\noindent
Second if $B$ is not a child of $A$, let $U$ be the first common parent of $A$ and $B$ and $C_A$ and $C_B$ the direct children of $U$ that are parents of $A$ and $B$ respectively. Without loss of generality assume that in the loop on lines \ref{state:forLoop} to \ref{state:endForLoop} $C_A$ is visited before $C_B$. Then $C_A$ and all of its children will appear in $L_1$ before $C_B$ its children. For $L_2$ the opposite is the case. Hence $I_1(A) < I_1(B)$ and $I_2(A) > I_2(B)$.
\end{proof}
